07 / Zero-copy-paste mobile log sync & SQLite architecture

Log daily on your phone in Google Docs — never copy-paste into Notepad again.

1. Direct Google Docs Cloud Pull (Zero Copy-Paste)

Keep logging daily on your phone in the Google Docs app inside your Logs folder. Link each Google Doc URL once on any Athletics/Health page (no Admin needed) or via CLI:

python -B -m site_content.log_sync --set-doc weightlifting "https://docs.google.com/document/d/..."
python -B -m site_content.log_sync

log_sync downloads the live plain text straight from Google Docs (or extracts .docx via docx_to_txt), repairs any paste/typo slips for 0 tokens, and updates both records/ and LiamMs_PandasProjects/.

2. How SQLite Works on Static GitHub Pages (Limits & Hybrid Sync)

records/workbench/workbench.db is a true relational SQLite 3 database (up to 281 TB theoretical capacity, ACID transactions, zero-config single file). Because GitHub Pages serves static files without a persistent write-disk server, live visitor clicks and mobile form entries stage immediately in browser localStorage and merge with the canonical SQLite database on each build or export.

CREATE TABLE IF NOT EXISTS study_queue (
    id TEXT PRIMARY KEY,
    created_on TEXT NOT NULL,
    title TEXT NOT NULL,
    atlas_path TEXT NOT NULL,
    question TEXT NOT NULL,
    dataset TEXT NOT NULL,
    status TEXT NOT NULL CHECK (status IN ('published', 'in-progress', 'queued')),
    study_route TEXT,
    paper_route TEXT,
    clicks INTEGER NOT NULL DEFAULT 0,
    notes TEXT NOT NULL DEFAULT ''
);

CREATE TABLE IF NOT EXISTS theory_queue (
    id TEXT PRIMARY KEY,
    created_on TEXT NOT NULL,
    category TEXT NOT NULL,
    title TEXT NOT NULL,
    atlas_path TEXT NOT NULL,
    scope_prompt TEXT NOT NULL,
    status TEXT NOT NULL CHECK (status IN ('published', 'in-progress', 'queued')),
    notes TEXT NOT NULL DEFAULT ''
);

CREATE TABLE IF NOT EXISTS applications_queue (
    id TEXT PRIMARY KEY,
    created_on TEXT NOT NULL,
    domain TEXT NOT NULL,
    title TEXT NOT NULL,
    atlas_path TEXT NOT NULL,
    system_design TEXT NOT NULL,
    societal_benefit TEXT NOT NULL,
    status TEXT NOT NULL CHECK (status IN ('published', 'in-progress', 'queued')),
    notes TEXT NOT NULL DEFAULT ''
);

CREATE TABLE IF NOT EXISTS wishlist_todos (
    id TEXT PRIMARY KEY,
    created_on TEXT NOT NULL,
    title TEXT NOT NULL,
    category TEXT NOT NULL,
    priority TEXT NOT NULL CHECK (priority IN ('high', 'medium', 'low')),
    atlas_path TEXT NOT NULL,
    status TEXT NOT NULL CHECK (status IN ('done', 'in-progress', 'planned')),
    notes TEXT NOT NULL DEFAULT ''
);

CREATE TABLE IF NOT EXISTS rap_lyrics (
    id TEXT PRIMARY KEY,
    created_on TEXT NOT NULL,
    bucket TEXT NOT NULL,
    title TEXT NOT NULL,
    bars TEXT NOT NULL,
    theme_cue TEXT NOT NULL DEFAULT '',
    status TEXT NOT NULL CHECK (status IN ('draft', 'recorded', 'published'))
);

CREATE TABLE IF NOT EXISTS voice_notes (
    id TEXT PRIMARY KEY,
    created_on TEXT NOT NULL,
    channel TEXT NOT NULL CHECK (channel IN ('Journal', 'Podcast', 'Study Memo', 'Freestyle')),
    title TEXT NOT NULL,
    duration_seconds INTEGER NOT NULL CHECK (duration_seconds >= 0),
    audio_format TEXT NOT NULL,
    transcript TEXT NOT NULL,
    summary TEXT NOT NULL,
    atlas_path TEXT NOT NULL,
    status TEXT NOT NULL CHECK (status IN ('processed', 'queued'))
);

CREATE TABLE IF NOT EXISTS site_traffic (
    ip_index TEXT PRIMARY KEY,
    masked_ip TEXT NOT NULL,
    role TEXT NOT NULL,
    visits INTEGER NOT NULL DEFAULT 1,
    study_clicks INTEGER NOT NULL DEFAULT 0,
    top_section TEXT NOT NULL,
    last_seen TEXT NOT NULL
);
